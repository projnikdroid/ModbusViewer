function(modbusviewer_add_test test_name)
    set(sources ${ARGN})
    add_executable(${test_name} ${sources})
    target_link_libraries(${test_name} PRIVATE modbusviewer_core Qt6::Test)
    add_test(NAME ${test_name} COMMAND ${test_name})
endfunction()

# For tests that exercise app-lib/ classes: compiles those sources into the test
# target (see the note in the top-level CMakeLists.txt) and adds the Qml dependency
# their QML_ELEMENT annotations pull in.
function(modbusviewer_add_applib_test test_name)
    set(sources ${ARGN})
    add_executable(${test_name} ${sources} ${MODBUSVIEWER_APPLIB_SOURCES})
    target_include_directories(${test_name} PRIVATE ${MODBUSVIEWER_APPLIB_INCLUDE_DIRS})
    target_link_libraries(${test_name} PRIVATE modbusviewer_core Qt6::Test Qt6::Qml Qt6::SerialPort)
    add_test(NAME ${test_name} COMMAND ${test_name})
endfunction()
